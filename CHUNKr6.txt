Command:
  python3 -c '
  from PIL import Image, ImageStat
  def interior(path, y0, y1, inset=0.22):
      im = Image.open(path).convert("RGB")
      w, _ = im.size
      crop = im.crop((int(w*inset), y0, int(w*(1-inset)), y1))
      stat = ImageStat.Stat(crop)
      print(path, round(sum(stat.var)/3, 2), [round(m,1) for m in stat.mean])
  interior("docs/screens/rc2/f5-session-initialized-390.png", 860, 1240)
  interior("docs/screens/rc2/f3-task-card-390.png", 980, 1340)
  '
